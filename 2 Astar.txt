class Astar{
    public static void main(String []args){
        int graph[][] = {
            {0, 2, 4, 0},
            {2, 0, 1, 7},
            {4, 1, 0, 3},
            {0, 7, 3, 0}
        };
        int h[]={7,6,2,0};
        int n=4;
        int start=0;
        int goal=3;
        boolean visit[]=new boolean[n];
        int current=start;
        System.out.println("Path:"+current);
        while(current!=goal){
            visit[current]=true;
            int min=999, next=-1;
            for(int i=0;i<n;i++){
                if(graph[current][i]!=0 && !visit[i]){
                    int f=graph[current][i]+h[i];
                    System.out.println("Node"+i+":"+f);
                    if(f<min){
                        min=f;
                        next=i;
                    }
                }
            }
            current=next;
            System.out.println("Selected node:"+current);
        }
        System.out.println("Destination reached");
    }
}

//output
Path:0
Node1:8
Node2:6
Selected node:2
Node1:7
Node3:3
Selected node:3
Destination reached
